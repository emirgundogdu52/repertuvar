🔍 Veri Çekme (SELECT)

SELECT
Tüm onaylı eserleri getir
works tablosundan status=approved olan tüm eserleri ada göre sıralı getirir
SELECT id, name, composer, makam, tuning, status
FROM works
WHERE status = 'approved'
ORDER BY name ASC;
SELECT
Bekleyen (pending) eserleri getir
Onay bekleyen, üye tarafından önerilen eserleri listeler
SELECT id, name, composer, submitter_email, submitted_at
FROM works
WHERE status = 'pending'
ORDER BY submitted_at DESC;
SELECT
Belirli makamda eserleri getir
Örnek: Hicaz makamındaki tüm eserler
SELECT id, name, composer, makam
FROM works
WHERE makam = 'Hicaz'
  AND status = 'approved'
ORDER BY name;
SELECT
Güftede kelime ara
lyrics alanında geçen bir kelimeyi içeren eserleri bulur
SELECT id, name, composer
FROM works
WHERE lyrics ILIKE '%ayrılık%'
  AND status = 'approved';
SELECT
Tüm kullanıcıları ve rollerini getir
profiles tablosundaki kayıtlı üyeleri listeler
SELECT p.id, p.role, u.email, u.created_at
FROM profiles p
JOIN auth.users u ON p.id = u.id
ORDER BY p.role, u.email;
SELECT
Makam bazında eser sayısı
Her makamda kaç eser olduğunu öğrenmek için
SELECT makam, COUNT(*) AS adet
FROM works
WHERE status = 'approved'
  AND makam IS NOT NULL
GROUP BY makam
ORDER BY adet DESC;
SELECT
Akor bilgisi olan eserler
chords alanı dolu olan eserleri listeler
SELECT id, name, composer
FROM works
WHERE chords IS NOT NULL
  AND chords != ''
  AND status = 'approved';

✏️ Güncelleme (UPDATE)

UPDATE
Kullanıcı rolünü güncelle
Email adresine göre bir kullanıcının rolünü değiştirir (admin / editor / member)
UPDATE profiles
SET role = 'editor'
WHERE id = (
  SELECT id FROM auth.users
  WHERE email = 'kullanici@email.com'
);
UPDATE
Eseri onayla (pending → approved)
ID'si bilinen bir eserin durumunu onaylıya çeker
UPDATE works
SET status = 'approved'
WHERE id = 123; -- eser ID'sini değiştir
UPDATE
Tüm NULL status'ları approved yap
Status alanı boş kalan eski eserleri toplu olarak onaylar
UPDATE works
SET status = 'approved'
WHERE status IS NULL;

➕ Ekleme (INSERT)

INSERT
Yeni kullanıcıya profil oluştur
Kayıt olan ama profiles tablosunda kaydı olmayan kullanıcı için profil oluşturur
INSERT INTO profiles (id, role)
VALUES (
  (SELECT id FROM auth.users WHERE email = 'kullanici@email.com'),
  'member'
)
ON CONFLICT (id) DO NOTHING;

🗑 Silme (DELETE)

DELETE
ID'ye göre eser sil
Belirli bir eseri kalıcı olarak siler — geri alınamaz!
-- Önce kontrol et:
SELECT * FROM works WHERE id = 123;

-- Sonra sil:
DELETE FROM works WHERE id = 123;
DELETE
Tüm pending eserleri sil
Reddedilen veya temizlemek istediğin bekleyen önerileri siler
-- Önce kaç tane olduğunu gör:
SELECT COUNT(*) FROM works WHERE status = 'pending';

-- Sonra sil:
DELETE FROM works WHERE status = 'pending';

🔧 Tablo Değişiklikleri (ALTER)

ALTER
Tabloya yeni kolon ekle
works tablosuna yeni bir text alanı ekler (örnek: chords)
ALTER TABLE works
ADD COLUMN yeni_kolon text;
ALTER
Kolonu sil
Tablodaki bir kolonu tamamen kaldırır — geri alınamaz!
ALTER TABLE works
DROP COLUMN kolon_adi;

🛠 Yardımcı Sorgular

SELECT
Tüm kolonları gör
Bir tablonun hangi kolonları olduğunu öğrenmek için
SELECT column_name, data_type, is_nullable
FROM information_schema.columns
WHERE table_name = 'works'
ORDER BY ordinal_position;
SELECT
Tekrar eden eser adlarını bul
Aynı isimde birden fazla eser var mı kontrol eder
SELECT name, COUNT(*) AS adet
FROM works
WHERE status = 'approved'
GROUP BY name
HAVING COUNT(*) > 1
ORDER BY adet DESC;
SELECT
Boş güfteli eserleri bul
Lyrics alanı boş olan eserleri listeler
SELECT id, name, composer
FROM works
WHERE (lyrics IS NULL OR lyrics = '')
  AND status = 'approved'
ORDER BY name;
SELECT
Belirli kullanıcının UUID'sini bul
Email adresine göre kullanıcı ID'sini öğrenmek için
SELECT id, email, created_at
FROM auth.users
WHERE email = 'kullanici@email.com';
